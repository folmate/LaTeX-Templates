  \setcounter{responseChangeListCounter}{0}%

\newcommand{\changeItem}[2][]{%
  \stepcounter{responseChangeListCounter}%
  \refstepcounter{responseChangeCounter}%
  \def\@tempa{#1}%
  \ifstrempty{\@tempa}%
    {\item \textbf{\myenumkey\arabic{responseChangeListCounter}}: #2}%
    {\item \textbf{\myenumkey\arabic{responseChangeListCounter}~#1:} #2}%
}
