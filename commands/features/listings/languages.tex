\usepackage{xparse} % needed for easier optional arguments

\newcommand*{\langKeywordStyle}[1]{{\normalfont \color{keywordcolor}\textbf{#1}}}
\NewDocumentCommand{\langkeyword}{m o}{%
    % #1 text to print and default command name
    % #2 (optional) override command name
    \IfNoValueTF{#2}{%
        \expandafter\newcommand\csname #1\endcsname{\langKeywordStyle{#1}}%
    }{%
        \expandafter\newcommand\csname #2\endcsname{\langKeywordStyle{#1}}%
    }%
}
